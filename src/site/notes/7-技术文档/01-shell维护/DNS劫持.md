## 4.2 其他脚本

/etc/haproxy/batch.sh

```bash
#!/bin/bash

FILE="list"
LINE=`cat ${FILE} | wc -l`

for (( I=1; I<=${LINE}; I++ )); do
  DOMAIN=`cat ${FILE} | sed -n "${I}p" | cut -d"," -f1`
  IP=`cat ${FILE} | sed -n "${I}p" | cut -d"," -f2`
  METHOD=`cat ${FILE} | sed -n "${I}p" | cut -d"," -f3`

  if [[ ${METHOD} == "https" ]]; then
    echo "# 前端使用"
    echo "  acl ACL_${DOMAIN} req_ssl_sni -i ${DOMAIN}"
    echo "  use_backend BACKEND_HTTPS_${DOMAIN} if ACL_${DOMAIN}"
    echo "# 后端使用"
    echo "  backend BACKEND_HTTPS_${DOMAIN}"
    echo "   mode tcp"
    echo "   server ${DOMAIN}_${IP} ${IP}:443 check check-ssl verify none"
    echo ""
  else
    echo "# 前端使用"
    echo "  acl ACL_${DOMAIN} hdr(host) -i ${DOMAIN}"
    echo "  use_backend BACKEND_HTTP_${DOMAIN} if ACL_${DOMAIN}"
    echo "# 后端使用"
    echo "  backend BACKEND_HTTP_${DOMAIN}"
    echo "    mode http"
    echo "    server ${DOMAIN}_${IP} ${IP}:80 check"
    echo ""
  fi
done
